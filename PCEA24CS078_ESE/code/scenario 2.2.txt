Step 2: Display only Electronics orders using $match and $project

db.orders.aggregate([
  { $match: { category: "Electronics" } },
  {
    $project: {
      _id: 0,
      orderId: 1,
      product: 1
    }
  }
])

Step 3: Calculate line total using $multiply

db.orders.aggregate([
  {
    $project: {
      _id: 0,
      orderId: 1,
      customer: 1,
      lineTotal: { $multiply: ["$qty", "$price"] }
    }
  }
])

Step 4: Find total quantity sold in each category using $group and $sum

db.orders.aggregate([
  {
    $group: {
      _id: "$category",
      totalQty: { $sum: "$qty" }
    }
  }
])


Step 5: Calculate revenue and average price using $sum, $avg and $sort

db.orders.aggregate([
  {
    $group: {
      _id: "$category",
      revenue: {
        $sum: { $multiply: ["$qty", "$price"] }
      },
      avgPrice: { $avg: "$price" }
    }
  },
  { $sort: { revenue: -1 } }
])

Step 6: Find top two customers by total spending using $group, $sort and $limit

db.orders.aggregate([
  {
    $group: {
      _id: "$customer",
      spent: {
        $sum: { $multiply: ["$qty", "$price"] }
      }
    }
  },
  { $sort: { spent: -1 } },
  { $limit: 2 }
])
